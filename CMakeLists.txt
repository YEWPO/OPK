cmake_minimum_required(VERSION 3.24)
project(OpenKeysPiano VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
include(CTest)
find_package(Qt6 6.5 REQUIRED COMPONENTS Core Widgets Concurrent)
if(BUILD_TESTING)
  find_package(Qt6 6.5 REQUIRED COMPONENTS Test)
endif()
if(WIN32)
  find_package(FluidSynth CONFIG REQUIRED)
  find_package(portaudio CONFIG REQUIRED)
  set(SYNTH_TARGET FluidSynth::libfluidsynth)
  if(TARGET PortAudio::PortAudio)
    set(AUDIO_TARGET PortAudio::PortAudio)
  elseif(TARGET portaudio)
    set(AUDIO_TARGET portaudio)
  elseif(TARGET portaudio_static)
    set(AUDIO_TARGET portaudio_static)
  else()
    message(FATAL_ERROR "Unsupported PortAudio CMake package")
  endif()
else()
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(FLUID REQUIRED IMPORTED_TARGET fluidsynth)
  pkg_check_modules(PORTAUDIO REQUIRED IMPORTED_TARGET portaudio-2.0)
  set(SYNTH_TARGET PkgConfig::FLUID)
  set(AUDIO_TARGET PkgConfig::PORTAUDIO)
endif()
add_library(piano_core src/core/model.cpp src/core/song_io.cpp src/core/eop.cpp src/core/keyboard.cpp)
target_include_directories(piano_core PUBLIC src)
target_link_libraries(piano_core PUBLIC Qt6::Core)
add_library(piano_audio src/audio/engine.cpp src/audio/soundfont_import.cpp)
target_link_libraries(piano_audio PUBLIC piano_core PRIVATE ${SYNTH_TARGET} ${AUDIO_TARGET})
add_library(piano_diagnostics src/audio/soundfont_check.cpp)
target_link_libraries(piano_diagnostics PUBLIC piano_audio)
add_executable(openkeys-soundfont-check src/soundfont_check_main.cpp)
target_link_libraries(openkeys-soundfont-check PRIVATE piano_diagnostics)
qt_add_resources(openkeys-soundfont-check check_assets PREFIX "/" FILES assets/demo.eopiano.json)
add_library(piano_ui src/ui/main_window.cpp src/ui/keyboard_view.cpp src/platform/input.cpp)
target_link_libraries(piano_ui PUBLIC piano_audio Qt6::Widgets PRIVATE Qt6::Concurrent)
qt_add_resources(piano_ui import_guide PREFIX "/" FILES docs/soundfont.md assets/demo.eopiano.json)
qt_add_executable(openkeys src/main.cpp)
target_link_libraries(openkeys PRIVATE piano_ui Qt6::Concurrent)
if(WIN32)
  target_link_libraries(piano_ui PRIVATE user32)
  set_target_properties(openkeys PROPERTIES WIN32_EXECUTABLE TRUE)
  target_compile_definitions(openkeys PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
endif()
foreach(t piano_core piano_audio piano_diagnostics piano_ui openkeys openkeys-soundfont-check)
  if(WIN32)
    target_compile_definitions(${t} PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  endif()
  if(MSVC)
    target_compile_options(${t} PRIVATE /W4 /utf-8 /permissive-)
  else()
    target_compile_options(${t} PRIVATE -Wall -Wextra -Wpedantic)
  endif()
endforeach()
if(BUILD_TESTING)
  add_executable(core_tests tests/core_tests.cpp)
  target_link_libraries(core_tests PRIVATE piano_core Qt6::Test)
  add_test(NAME core COMMAND core_tests)
  add_executable(soundfont_import_tests tests/soundfont_import_tests.cpp)
  target_link_libraries(soundfont_import_tests PRIVATE piano_audio Qt6::Test)
  add_test(NAME soundfont_import COMMAND soundfont_import_tests)
  set_tests_properties(soundfont_import PROPERTIES TIMEOUT 30)
  add_executable(audio_tests tests/audio_tests.cpp)
  target_link_libraries(audio_tests PRIVATE piano_audio Qt6::Test)
  add_test(NAME audio COMMAND audio_tests)
  set_tests_properties(core audio PROPERTIES TIMEOUT 30)
  add_executable(ui_tests tests/ui_tests.cpp)
  target_link_libraries(ui_tests PRIVATE piano_ui Qt6::Concurrent Qt6::Test)
  add_test(NAME ui COMMAND ui_tests)
  set_tests_properties(ui PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 20)
  add_executable(soundfont_check_tests tests/soundfont_check_tests.cpp)
  target_link_libraries(soundfont_check_tests PRIVATE piano_diagnostics Qt6::Test)
  add_dependencies(soundfont_check_tests openkeys-soundfont-check)
  add_test(NAME soundfont_check COMMAND soundfont_check_tests)
  set_tests_properties(soundfont_check PROPERTIES TIMEOUT 60)
  add_executable(real_soundfont_tests tests/real_soundfont_tests.cpp)
  target_link_libraries(real_soundfont_tests PRIVATE Qt6::Core)
  add_dependencies(real_soundfont_tests openkeys-soundfont-check)
  add_test(NAME real_soundfont COMMAND real_soundfont_tests)
  set_tests_properties(real_soundfont PROPERTIES SKIP_RETURN_CODE 77 TIMEOUT 360 LABELS real_soundfont)
  if(MSVC)
    foreach(t core_tests audio_tests ui_tests soundfont_check_tests real_soundfont_tests soundfont_import_tests)
      target_compile_options(${t} PRIVATE /utf-8)
    endforeach()
  endif()
  add_test(NAME app_smoke COMMAND openkeys --smoke-test)
  set_tests_properties(app_smoke PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 20)
  if(WIN32 AND DEFINED VCPKG_INSTALLED_DIR AND DEFINED VCPKG_TARGET_TRIPLET)
    set(_test_dependencies "${VCPKG_INSTALLED_DIR}/${VCPKG_TARGET_TRIPLET}")
    set_tests_properties(core soundfont_import audio ui soundfont_check real_soundfont app_smoke
      PROPERTIES ENVIRONMENT_MODIFICATION
      "PATH=path_list_prepend:${_test_dependencies}/bin;QT_PLUGIN_PATH=path_list_prepend:${_test_dependencies}/Qt6/plugins")
  endif()
endif()
if(NOT WIN32)
  install(TARGETS openkeys openkeys-soundfont-check RUNTIME DESTINATION .)
endif()
install(FILES LICENSE THIRD_PARTY_NOTICES.md DESTINATION licenses)
install(DIRECTORY licenses/ DESTINATION licenses)
install(FILES README.md DESTINATION .)
install(DIRECTORY docs/ DESTINATION docs FILES_MATCHING PATTERN "*.md" PATTERN "*.png")
install(FILES assets/demo.eopiano.json DESTINATION examples)
if(WIN32)
  install(TARGETS openkeys openkeys-soundfont-check RUNTIME_DEPENDENCIES
    DIRECTORIES "$<TARGET_FILE_DIR:${SYNTH_TARGET}>" "$<TARGET_FILE_DIR:${AUDIO_TARGET}>"
    PRE_EXCLUDE_REGEXES "api-ms-.*" "ext-ms-.*"
    POST_EXCLUDE_REGEXES ".*[Ww]indows[/\\\\][Ss]ystem32.*"
    RUNTIME DESTINATION .)
  qt_generate_deploy_app_script(TARGET openkeys OUTPUT_SCRIPT qt_deploy NO_UNSUPPORTED_PLATFORM_ERROR)
  # Executables and DLLs are installed at the package root.
  install(CODE [[set(QT_DEPLOY_BIN_DIR ".")]])
  install(SCRIPT ${qt_deploy})
  if(DEFINED VCPKG_INSTALLED_DIR AND DEFINED VCPKG_TARGET_TRIPLET)
    install(DIRECTORY "${VCPKG_INSTALLED_DIR}/${VCPKG_TARGET_TRIPLET}/share/" DESTINATION licenses/dependencies
      FILES_MATCHING PATTERN copyright PATTERN "LICENSE*" PATTERN "COPYING*")
  endif()
endif()
set(CPACK_PACKAGE_NAME OpenKeysPiano)
set(CPACK_PACKAGE_FILE_NAME "OpenKeysPiano-${PROJECT_VERSION}-${CMAKE_SYSTEM_NAME}-${CMAKE_SYSTEM_PROCESSOR}")
set(CPACK_PACKAGE_VENDOR OpenKeysPiano)
set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "Offline computer-keyboard piano")
set(CPACK_PACKAGE_INSTALL_DIRECTORY OpenKeysPiano)
set(CPACK_RESOURCE_FILE_LICENSE "${CMAKE_SOURCE_DIR}/LICENSE")
set(CPACK_PACKAGE_EXECUTABLES openkeys "OpenKeys Piano")
if(WIN32)
  set(CPACK_GENERATOR "NSIS;ZIP")
  set(CPACK_NSIS_DISPLAY_NAME "OpenKeys Piano")
  set(CPACK_NSIS_ENABLE_UNINSTALL_BEFORE_INSTALL ON)
else()
  set(CPACK_GENERATOR TGZ)
endif()
include(CPack)
