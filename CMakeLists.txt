cmake_minimum_required(VERSION 3.24)
project(OpenKeysPiano VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
include(CTest)
find_package(Qt6 6.5 REQUIRED COMPONENTS Core Widgets Concurrent LinguistTools)
if(BUILD_TESTING)
  find_package(Qt6 6.5 REQUIRED COMPONENTS Test)
endif()
if(WIN32)
  find_package(FluidSynth CONFIG REQUIRED)
  find_package(portaudio CONFIG REQUIRED)
  set(SYNTH_TARGET FluidSynth::libfluidsynth)
  if(TARGET PortAudio::PortAudio)
    set(AUDIO_TARGET PortAudio::PortAudio)
  elseif(TARGET portaudio)
    set(AUDIO_TARGET portaudio)
  elseif(TARGET portaudio_static)
    set(AUDIO_TARGET portaudio_static)
  else()
    message(FATAL_ERROR "Unsupported PortAudio CMake package")
  endif()
else()
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(FLUID REQUIRED IMPORTED_TARGET fluidsynth)
  pkg_check_modules(PORTAUDIO REQUIRED IMPORTED_TARGET portaudio-2.0)
  set(SYNTH_TARGET PkgConfig::FLUID)
  set(AUDIO_TARGET PkgConfig::PORTAUDIO)
endif()
add_library(piano_core src/core/model.cpp src/core/song_io.cpp src/core/eop.cpp src/core/keyboard.cpp src/core/language.cpp)
target_include_directories(piano_core PUBLIC src)
target_link_libraries(piano_core PUBLIC Qt6::Core)
qt_add_lrelease(piano_core TS_FILES translations/openkeys_zh_CN.ts
  QM_FILES_OUTPUT_VARIABLE app_translations)
# Embed Qt's standard button/dialog translations as well, including in portable packages.
find_file(qt_chinese_translation qtbase_zh_CN.qm
  HINTS "${QT6_INSTALL_PREFIX}/${QT6_INSTALL_TRANSLATIONS}"
    "${Qt6_DIR}/../../translations" "${Qt6_DIR}/../../Qt6/translations"
    "${Qt6_DIR}/../../../translations" "${Qt6_DIR}/../../../share/qt6/translations"
    "/usr/share/qt6/translations" REQUIRED)
set_source_files_properties("${qt_chinese_translation}" PROPERTIES QT_RESOURCE_ALIAS qtbase_zh_CN.qm)
set_source_files_properties(${app_translations} PROPERTIES QT_RESOURCE_ALIAS openkeys_zh_CN.qm)
qt_add_resources(piano_core translations PREFIX "/i18n" FILES
  ${app_translations} "${qt_chinese_translation}")
add_library(piano_audio src/audio/engine.cpp src/audio/soundfont_import.cpp)
target_link_libraries(piano_audio PUBLIC piano_core PRIVATE ${SYNTH_TARGET} ${AUDIO_TARGET})
add_library(piano_diagnostics src/audio/soundfont_check.cpp)
target_link_libraries(piano_diagnostics PUBLIC piano_audio)
add_executable(openkeys-soundfont-check src/soundfont_check_main.cpp)
target_link_libraries(openkeys-soundfont-check PRIVATE piano_diagnostics)
qt_add_resources(openkeys-soundfont-check check_assets PREFIX "/" FILES assets/demo.eopiano.json)
add_library(piano_ui src/ui/main_window.cpp src/ui/keyboard_view.cpp src/platform/input.cpp)
target_link_libraries(piano_ui PUBLIC piano_audio Qt6::Widgets PRIVATE Qt6::Concurrent)
qt_add_resources(piano_ui import_guide PREFIX "/" FILES docs/soundfont.md docs/soundfont.zh-CN.md assets/demo.eopiano.json)
qt_add_resources(piano_ui app_icons PREFIX "/" FILES
  assets/icons/openkeys-16.png assets/icons/openkeys-24.png assets/icons/openkeys-32.png
  assets/icons/openkeys-48.png assets/icons/openkeys-64.png assets/icons/openkeys-128.png
  assets/icons/openkeys-256.png)
qt_add_executable(openkeys src/main.cpp)
target_link_libraries(openkeys PRIVATE piano_ui Qt6::Concurrent)
if(WIN32)
  enable_language(RC)
  target_sources(openkeys PRIVATE src/platform/openkeys.rc)
  target_include_directories(openkeys PRIVATE "${CMAKE_SOURCE_DIR}/assets/icons")
  target_link_libraries(piano_ui PRIVATE user32)
  set_target_properties(openkeys PROPERTIES WIN32_EXECUTABLE TRUE)
  target_compile_definitions(openkeys PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
endif()
foreach(t piano_core piano_audio piano_diagnostics piano_ui openkeys openkeys-soundfont-check)
  if(WIN32)
    target_compile_definitions(${t} PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  endif()
  if(MSVC)
    target_compile_options(${t} PRIVATE /W4 /utf-8 /permissive-)
  else()
    target_compile_options(${t} PRIVATE -Wall -Wextra -Wpedantic)
  endif()
endforeach()
if(BUILD_TESTING)
  # Test executables follow the tests/<target>.cpp naming convention.
  function(piano_test target test_name library)
    add_executable(${target} tests/${target}.cpp)
    target_link_libraries(${target} PRIVATE ${library} ${ARGN})
    add_test(NAME ${test_name} COMMAND ${target})
    if(MSVC)
      target_compile_options(${target} PRIVATE /utf-8)
    endif()
  endfunction()

  piano_test(language_tests language piano_core Qt6::Test)
  piano_test(core_tests core piano_core Qt6::Test)
  piano_test(soundfont_import_tests soundfont_import piano_audio Qt6::Test)
  piano_test(audio_tests audio piano_audio Qt6::Test)
  piano_test(ui_tests ui piano_ui Qt6::Concurrent Qt6::Test)
  piano_test(soundfont_check_tests soundfont_check piano_diagnostics Qt6::Test)
  add_dependencies(soundfont_check_tests openkeys-soundfont-check)
  piano_test(real_soundfont_tests real_soundfont Qt6::Core)
  add_dependencies(real_soundfont_tests openkeys-soundfont-check)
  set_tests_properties(core audio soundfont_import PROPERTIES TIMEOUT 30)
  set_tests_properties(ui PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 20)
  set_tests_properties(soundfont_check PROPERTIES TIMEOUT 60)
  set_tests_properties(real_soundfont PROPERTIES SKIP_RETURN_CODE 77 TIMEOUT 360 LABELS real_soundfont)

  add_test(NAME app_smoke COMMAND openkeys --smoke-test)
  add_test(NAME app_smoke_zh COMMAND openkeys --smoke-test --language zh_CN)
  add_test(NAME app_invalid_language COMMAND openkeys --smoke-test --language invalid)
  set_tests_properties(app_smoke app_smoke_zh app_invalid_language PROPERTIES
    ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 20)
  set_tests_properties(app_invalid_language PROPERTIES WILL_FAIL TRUE)
  if(WIN32 AND DEFINED VCPKG_INSTALLED_DIR AND DEFINED VCPKG_TARGET_TRIPLET)
    set(_test_dependencies "${VCPKG_INSTALLED_DIR}/${VCPKG_TARGET_TRIPLET}")
    set_tests_properties(language core soundfont_import audio ui soundfont_check real_soundfont
      app_smoke app_smoke_zh app_invalid_language
      PROPERTIES ENVIRONMENT_MODIFICATION
      "PATH=path_list_prepend:${_test_dependencies}/bin;QT_PLUGIN_PATH=path_list_prepend:${_test_dependencies}/Qt6/plugins")
  endif()
endif()
if(NOT WIN32)
  install(TARGETS openkeys openkeys-soundfont-check RUNTIME DESTINATION .)
endif()
install(FILES LICENSE THIRD_PARTY_NOTICES.md THIRD_PARTY_NOTICES.zh-CN.md DESTINATION licenses)
install(FILES cmake/installer/CMake-LICENSE.rst DESTINATION licenses)
install(DIRECTORY licenses/ DESTINATION licenses)
install(FILES README.md README.zh-CN.md LICENSE THIRD_PARTY_NOTICES.md THIRD_PARTY_NOTICES.zh-CN.md DESTINATION .)
install(DIRECTORY docs/ DESTINATION docs FILES_MATCHING PATTERN "*.md" PATTERN "*.png")
install(FILES assets/demo.eopiano.json DESTINATION examples)
if(WIN32)
  install(TARGETS openkeys openkeys-soundfont-check RUNTIME_DEPENDENCIES
    DIRECTORIES "$<TARGET_FILE_DIR:${SYNTH_TARGET}>" "$<TARGET_FILE_DIR:${AUDIO_TARGET}>"
    PRE_EXCLUDE_REGEXES "api-ms-.*" "ext-ms-.*"
    POST_EXCLUDE_REGEXES ".*[Ww]indows[/\\\\][Ss]ystem32.*"
    RUNTIME DESTINATION .)
  qt_generate_deploy_app_script(TARGET openkeys OUTPUT_SCRIPT qt_deploy NO_UNSUPPORTED_PLATFORM_ERROR)
  # Executables and DLLs are installed at the package root.
  install(CODE [[set(QT_DEPLOY_BIN_DIR ".")]])
  install(SCRIPT ${qt_deploy})
  if(DEFINED VCPKG_INSTALLED_DIR AND DEFINED VCPKG_TARGET_TRIPLET)
    install(DIRECTORY "${VCPKG_INSTALLED_DIR}/${VCPKG_TARGET_TRIPLET}/share/" DESTINATION licenses/dependencies
      FILES_MATCHING PATTERN copyright PATTERN "LICENSE*" PATTERN "COPYING*")
  endif()
endif()

  set(CPACK_GENERATOR "NSIS;ZIP")
  set(CPACK_NSIS_DISPLAY_NAME "OpenKeys Piano")
  set(CPACK_NSIS_EXECUTABLES_DIRECTORY ".")
  set(CPACK_CREATE_DESKTOP_LINKS openkeys)

  # Let CPack quote/convert this path; TO_NATIVE_PATH applies shell escaping too early.

  set(CPACK_NSIS_MUI_ICON "${_installer_icon}")
  set(CPACK_NSIS_MUI_UNIICON "${_installer_icon}")
  # Repository-owned bilingual shortcut page, without PATH or Start Menu folder pages.
  list(PREPEND CMAKE_MODULE_PATH "${CMAKE_SOURCE_DIR}/cmake/installer")
  set(CPACK_NSIS_MODIFY_PATH OFF)
  set(CPACK_NSIS_DEFINES "Unicode true")
  set(CPACK_NSIS_ENABLE_UNINSTALL_BEFORE_INSTALL ON)
else()
  set(CPACK_GENERATOR TGZ)
endif()
